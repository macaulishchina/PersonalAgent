cmake_minimum_required(VERSION 3.16)
project(ppdemo_host C)
enable_testing()
add_executable(test_lab_model test_lab_model.c ../main/lab_model.c)
target_include_directories(test_lab_model PRIVATE ../main)
target_compile_features(test_lab_model PRIVATE c_std_11)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(test_lab_model PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME lab_model COMMAND test_lab_model)
