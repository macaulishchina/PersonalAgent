cmake_minimum_required(VERSION 3.16)
project(ppdemo_host C)
enable_testing()
add_executable(test_cache_store test_cache_store.c ../apps/pa_terminal/main/cache_store.c)
target_include_directories(test_cache_store PRIVATE ../apps/pa_terminal/main)
target_compile_definitions(test_cache_store PRIVATE _POSIX_C_SOURCE=200809L)
target_compile_options(test_cache_store PRIVATE -Wall -Wextra -Werror)
add_test(NAME cache_store COMMAND test_cache_store)
add_executable(test_lab_model test_lab_model.c ../main/lab_model.c)
target_include_directories(test_lab_model PRIVATE ../main)
target_compile_features(test_lab_model PRIVATE c_std_11)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(test_lab_model PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME lab_model COMMAND test_lab_model)

foreach(language en zh)
    add_executable(test_language_${language} test_language.c ../main/lab_language.c ../main/lab_model.c)
    target_include_directories(test_language_${language} PRIVATE ../main)
    target_compile_features(test_language_${language} PRIVATE c_std_11)
    if(language STREQUAL "zh")
        target_compile_definitions(test_language_${language} PRIVATE PPDEMO_ZH=1)
    endif()
    add_test(NAME language_${language} COMMAND test_language_${language})
endforeach()

# Explicit hardware-toolchain check: use its pinned cJSON instead of vendoring a copy.
set(PP_CJSON_SOURCE "" CACHE FILEPATH "ESP-IDF 5.5.3 components/json/cJSON/cJSON.c")
if(PP_CJSON_SOURCE)
    if(NOT EXISTS "${PP_CJSON_SOURCE}")
        message(FATAL_ERROR "PP_CJSON_SOURCE must name the pinned SDK cJSON source")
    endif()
    get_filename_component(PP_CJSON_DIRECTORY "${PP_CJSON_SOURCE}" DIRECTORY)
    add_executable(test_settings_model test_settings_model.c ../apps/pa_terminal/main/settings_model.c
                   ../apps/pa_terminal/main/network_model.c "${PP_CJSON_SOURCE}")
    target_include_directories(test_settings_model PRIVATE ../apps/pa_terminal/main "${PP_CJSON_DIRECTORY}")
    target_compile_options(test_settings_model PRIVATE -Wall -Wextra -Werror)
    if(NOT WIN32)
        target_link_libraries(test_settings_model PRIVATE m)
    endif()
    add_test(NAME settings_model COMMAND test_settings_model)
    add_executable(test_notification_model test_notification_model.c ../apps/pa_terminal/main/notification_model.c "${PP_CJSON_SOURCE}")
    target_include_directories(test_notification_model PRIVATE ../apps/pa_terminal/main "${PP_CJSON_DIRECTORY}")
    target_compile_options(test_notification_model PRIVATE -Wall -Wextra -Werror)
    add_test(NAME notification_model COMMAND test_notification_model)
    add_executable(test_timeline test_timeline.c ../apps/pa_terminal/main/timeline.c "${PP_CJSON_SOURCE}")
    target_include_directories(test_timeline PRIVATE ../apps/pa_terminal/main "${PP_CJSON_DIRECTORY}")
    target_compile_options(test_timeline PRIVATE -Wall -Wextra -Werror)
    if(NOT WIN32)
        target_link_libraries(test_timeline PRIVATE m)
    endif()
    add_test(NAME timeline COMMAND test_timeline)
    add_executable(test_history_cache test_history_cache.c ../apps/pa_terminal/main/history_cache.c
                   ../apps/pa_terminal/main/timeline.c "${PP_CJSON_SOURCE}")

    target_compile_options(test_history_cache PRIVATE -Wall -Wextra -Werror)
    if(NOT WIN32)
        target_link_libraries(test_history_cache PRIVATE m)
    endif()
    add_test(NAME history_cache COMMAND test_history_cache)
endif()
