cmake_minimum_required(VERSION 3.16)
project(ppdemo_host C)
enable_testing()
add_executable(test_lab_model test_lab_model.c ../main/lab_model.c)
target_include_directories(test_lab_model PRIVATE ../main)
target_compile_features(test_lab_model PRIVATE c_std_11)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(test_lab_model PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME lab_model COMMAND test_lab_model)

foreach(language en zh)
    add_executable(test_language_${language} test_language.c ../main/lab_language.c ../main/lab_model.c)
    target_include_directories(test_language_${language} PRIVATE ../main)
    target_compile_features(test_language_${language} PRIVATE c_std_11)
    if(language STREQUAL "zh")
        target_compile_definitions(test_language_${language} PRIVATE PPDEMO_ZH=1)
    endif()
    add_test(NAME language_${language} COMMAND test_language_${language})
endforeach()

add_executable(test_terminal_model test_terminal_model.c ../apps/pa_terminal/main/terminal_model.c)
target_include_directories(test_terminal_model PRIVATE ../apps/pa_terminal/main)
target_compile_options(test_terminal_model PRIVATE -Wall -Wextra -Werror)
add_test(NAME terminal_model COMMAND test_terminal_model)
